using System;

class SavingsCalculator
{
    static void Main()
    {
        Console.WriteLine("=== Savings Goal Calculator ===\n");

        string goal = ReadNonEmptyString("What are you saving for? Enter title: ");

        double totalNeeded = ReadPositiveDouble("How much do you need to save? Enter dollars: ");
        double totalWeeks = ReadPositiveDouble("How long before the purchase? Enter weeks: ");
        double currentSavings = ReadNonNegativeDouble("How much do you have already? Enter dollars: ");
        double weeklySavings = ReadPositiveDouble("How much can you save each week? Enter dollars: ");

        if (weeklySavings == 0)
        {
            Console.WriteLine("Weekly savings must be greater than 0.");
            return;
        }

        double requiredWeeklySaving = (totalNeeded - currentSavings) / totalWeeks;
        double weeksNeeded = (totalNeeded - currentSavings) / weeklySavings;

        requiredWeeklySaving = Math.Max(0, requiredWeeklySaving);
        weeksNeeded = Math.Max(0, weeksNeeded);

        Console.WriteLine("\n=== Results ===");
        Console.WriteLine($"For the '{goal}', you need to save ${requiredWeeklySaving:F2} per week.");
        Console.WriteLine($"At ${weeklySavings:F2} per week, you will reach your goal in {weeksNeeded:F1} weeks.");
    }

    static string ReadNonEmptyString(string prompt)
    {
        while (true)
        {
            Console.Write(prompt);
            string input = Console.ReadLine();
            if (!string.IsNullOrWhiteSpace(input))
            {
                return input.Trim();
            }
            Console.WriteLine("Input cannot be empty. Please enter something.");
        }
    }

    static double ReadPositiveDouble(string prompt)
    {
        while (true)
        {
            Console.Write(prompt);
            string input = Console.ReadLine();
            if (double.TryParse(input, out double result) && result > 0)
            {
                return result;
            }
            Console.WriteLine("Invalid input. Please enter a number greater than 0.");
        }
    }

    static double ReadNonNegativeDouble(string prompt)
    {
        while (true)
        {
            Console.Write(prompt);
            string input = Console.ReadLine();
            if (double.TryParse(input, out double result) && result >= 0)
            {
                return result;
            }
            Console.WriteLine("Invalid input. Please enter a number 0 or greater.");
        }
    }
}
